function generateInvoice() { if (cart.length === 0) { showNotification('⚠️ Carrito vacío.', 'warning'); return; } try { const { jsPDF } = window.jspdf; const doc = new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' }); const pageWidth = doc.internal.pageSize.getWidth(); const margin = 15; const today = new Date(); const invoiceNumber = 'FAC-' + Math.floor(Math.random() * 10000).toString().padStart(4, '0'); const invoiceDate = today.toLocaleDateString('es-AR'); const invoiceTime = today.toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' }); const dueDate = new Date(today); dueDate.setDate(dueDate.getDate() + 30); const dueDateStr = dueDate.toLocaleDateString('es-AR'); const subtotal = cart.reduce((sum, item) => sum + item.price * item.quantity, 0); const tax = subtotal * (TAX_RATE / 100); const total = subtotal + tax; // --- Encabezado --- doc.setFillColor(37, 99, 235); doc.circle(20, 20, 8, 'F'); doc.setTextColor(255, 255, 255); doc.setFontSize(12); doc.text('📚', 16, 24); doc.setTextColor(0, 0, 0); doc.setFontSize(18); doc.setFont('helvetica', 'bold'); doc.text('Librería El Sabio', 40, 20); doc.setFontSize(8); doc.setFont('helvetica', 'normal'); doc.text('NIT: 12.345.678-9', 40, 26); doc.text('Calle del Conocimiento 123, Buenos Aires', 40, 31); doc.text('+54 11 4567-8901', 40, 36); doc.text('info@libreriaelsabio.com', 40, 41); // Datos factura doc.setFontSize(11); doc.setFont('helvetica', 'bold'); doc.text('FACTURA', pageWidth - margin, 20, { align: 'right' }); doc.setFontSize(8); doc.setFont('helvetica', 'normal'); doc.text(`No. ${invoiceNumber}`, pageWidth - margin, 26, { align: 'right' }); doc.text(`Fecha: ${invoiceDate}`, pageWidth - margin, 31, { align: 'right' }); doc.text(`Hora: ${invoiceTime}`, pageWidth - margin, 36, { align: 'right' }); doc.text(`Vencimiento: ${dueDateStr}`, pageWidth - margin, 41, { align: 'right' }); doc.setDrawColor(200); doc.line(margin, 48, pageWidth - margin, 48); // Cliente doc.setFontSize(10); doc.setFont('helvetica', 'bold'); doc.text('DATOS DEL CLIENTE', margin, 56); doc.setFontSize(8); doc.setFont('helvetica', 'normal'); doc.text('Cliente: Consumidor Final', margin, 62); doc.text('NIT / CC: 00-00000000-0', margin, 67); doc.text('Dirección: -', margin, 72); doc.text('Email: -', margin, 77); doc.text('Teléfono: -', margin, 82); // Tabla const headers = ['ÍTEM', 'DESCRIPCIÓN', 'CANT.', 'U. MEDIDA', 'PRECIO UNIT.', 'TOTAL']; const rows = cart.map((item, idx) => [ (idx + 1).toString(), `${item.title} - ${item.author}`, item.quantity.toString(), 'Und', `$ ${item.price.toFixed(2)}`, `$ ${(item.price * item.quantity).toFixed(2)}` ]); doc.autoTable({ startY: 90, margin: { left: margin, right: margin }, head: [headers], body: rows, theme: 'grid', styles: { fontSize: 7, cellPadding: 2 }, headStyles: { fillColor: [37, 99, 235], textColor: 255, fontStyle: 'bold', fontSize: 7 }, columnStyles: { 0: { cellWidth: 10, halign: 'center' }, 1: { cellWidth: 50 }, 2: { cellWidth: 12, halign: 'center' }, 3: { cellWidth: 18, halign: 'center' }, 4: { cellWidth: 26, halign: 'right' }, 5: { cellWidth: 26, halign: 'right' } } }); const finalY = doc.lastAutoTable.finalY + 8; // Observaciones doc.setFontSize(8); doc.setFont('helvetica', 'italic'); doc.text('Gracias por su preferencia.', margin, finalY); doc.text('Todos los libros cuentan con garantía de satisfacción.', margin, finalY + 5); doc.text('Para cambios o devoluciones, presente esta factura.', margin, finalY + 10); // Totales const totalsX = pageWidth - margin - 60; doc.text('SUBTOTAL', totalsX, finalY + 20); doc.text(`$ ${subtotal.toFixed(2)}`, pageWidth - margin, finalY + 20, { align: 'right' }); doc.text('DESCUENTO', totalsX, finalY + 27); doc.text('$ 0.00', pageWidth - margin, finalY + 27, { align: 'right' }); doc.text(`IVA (${TAX_RATE}%)`, totalsX, finalY + 34); doc.text(`$ ${tax.toFixed(2)}`, pageWidth - margin, finalY + 34, { align: 'right' }); doc.line(totalsX, finalY + 37, pageWidth - margin, finalY + 37); doc.setFont('helvetica', 'bold'); doc.setFontSize(10); doc.text('TOTAL A PAGAR', totalsX, finalY + 43); doc.text(`$ ${total.toFixed(2)}`, pageWidth - margin, finalY + 43, { align: 'right' }); // Métodos de pago doc.setFontSize(9); doc.text('MÉTODOS DE PAGO', margin, finalY + 55); doc.setFontSize(8); doc.setFont('helvetica', 'normal'); doc.text('• Transferencia: Banco Nación, Cuenta Corriente No. 123456789', margin, finalY + 62); doc.text('• Tarjeta: Débito y crédito (Visa, Mastercard)', margin, finalY + 68); doc.text('• Efectivo: Pago en nuestro local', margin, finalY + 74); doc.setFont('helvetica', 'italic'); doc.setTextColor(100); doc.text('¡Gracias por su compra!', pageWidth / 2, finalY + 85, { align: 'center' }); // Descargar la factura primero doc.save(`factura_${invoiceNumber}.pdf`); // Intentar guardar en el historial (no debe bloquear la factura) try { const invoiceData = { number: invoiceNumber, date: invoiceDate + ' ' + invoiceTime, items: cart.map(item => ({ ...item })), subtotal: subtotal, tax: tax, total: total }; invoiceHistory.push(invoiceData); saveInvoiceHistory(); updateHistoryUI(); showNotification('📄 Factura generada y guardada en el historial.', 'success'); } catch (histError) { console.warn('No se pudo guardar el historial:', histError); showNotification('📄 Factura generada, pero no se guardó en el historial (memoria llena).', 'warning'); } // Limpiar carrito setTimeout(() => { cart = []; updateCart(); saveCartToLocalStorage(); }, 2000); } catch (e) { console.error(e); showNotification('❌ Error al generar la factura.', 'error'); } }
